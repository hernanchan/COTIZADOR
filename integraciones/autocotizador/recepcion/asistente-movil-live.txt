 function fieldName(label){
  return String(label.childNodes[0]?.textContent||'').replace(/\s+/g,' ').trim().toLowerCase();

 function partFor(node){
  if(node.matches('label')){
   const t=fieldName(node);
   if(/^(impresión|material|papel|producto y variante|tipo de corte)/.test(t))return 3;
   if(/^(caras|páginas por cara)/.test(t))return 4;
   if(/^(copias|cantidad|anillado|grupo de anillado|metros lineales)/.test(t))return 5;
   return 2;
  }
  if(node.matches('.composicion-visible,.composicion-por-cara'))return 4;
  if(node.matches('.tamano-rapido,.pregunta-tamano,.eleccion-plotter,.selector-paginas-grande'))return 2;
  const t=String(node.textContent||'').slice(0,160).toLowerCase();
  if(t.includes('anillado')||t.includes('los pdf del grupo'))return 5;
  if(t.includes('doble faz')||t.includes('páginas por cara'))return 4;
  return 2;
 }

  }
  const panel=card.querySelector('.panel-edicion');
  if(!panel)return;
  for(const child of panel.children){
   if(child.matches('.controles')){
    let count=0;
    for(const field of child.children){
     const part=partFor(field);
     field.dataset.mobilePart=String(part);
     field.dataset.mobileHidden=String(part!==stage);
     if(part===stage)count++;
    }
    child.dataset.mobileHidden=String(count===0);
   }else{
    const part=partFor(child);
    child.dataset.mobilePart=String(part);
    child.dataset.mobileHidden=String(part!==stage);
   }
  }

   next.textContent=stage===2&&commonMode()&&activeFile<numFiles()-1?'Siguiente archivo':stage===2&&commonMode()?'Configurar opciones compartidas':stage===2?'Elegir color y papel':stage===3&&commonMode()?'Revisar presupuesto':stage===3?'Elegir caras':
    stage===4?'Copias y terminación':
     activeFile<numFiles()-1?'Siguiente archivo':'Revisar presupuesto';
   next.disabled=numFiles()===0;actions.append(prev,next);
   const card=cards()[activeFile];
   const controls=card?[...card.querySelectorAll('.panel-edicion .controles>[data-mobile-part],.composicion-visible[data-mobile-part]')]:[];
   if((stage===3||stage===4)&&card?.querySelector('.panel-edicion')&&
      !controls.some(x=>x.dataset.mobilePart===String(stage))){
    help.hidden=false;help.textContent='Este servicio no requiere otras opciones en este paso. Podés continuar.';
   }
   if(warning){help.hidden=false;help.textContent=warning;}
  }else{
   actions.style.gridTemplateColumns='1fr';
   fileCount.textContent=numFiles()+' archivo(s) · '+(price!==null?money(price):'Total pendiente');
   reviewRows();
   description.textContent=sendable&&price!==null
    ?'Presupuesto listo para revisar. En esta prueba no se genera ningún pedido.'
    :'Faltan opciones o cotización. Volvé a la configuración para revisarlas.';
   actions.append(edit);
  }
 }
 function go(n){
  if((n>=2&&n<=6)&&numFiles()<1)return;
  if(n<1||n>6)return;
  stage=n;warning='';
  if(n===1||n===6)list.querySelectorAll('.archivo,.grupo-pedido').forEach(x=>{x.hidden=false});
  refresh();postStage();
  nav.scrollIntoView({behavior:'instant',block:'start'});
 }
 function goBack(){
  if(stage===6){activeFile=Math.max(0,numFiles()-1);go(5);return}
  if(stage===3&&commonMode()){activeFile=Math.max(0,numFiles()-1);go(2);return}
  if(stage===2&&activeFile>0){activeFile--;go(5);return}
  go(stage-1);
 }
 function goForward(){
  if(stage===1){go(2);return}
  if(stage>=2&&stage<=5){
   warning=checkFile();
   if(warning){refresh();return}
   if(stage===2&&commonMode()&&activeFile<numFiles()-1){activeFile++;go(2);return;}
   if(stage===3&&commonMode()){go(6);return;}
   if(stage===5){
    if(activeFile<numFiles()-1){activeFile++;go(2);return}
    go(6);return;
   }
   go(stage+1);
  }
 }
 function enable(){
  if(active()){refresh();postStage()}
  else{
   actions.hidden=true;
   document.body.classList.remove('saber-mobile');
   document.body.removeAttribute('data-mobile-stage');
  document.body.removeAttribute('data-mobile-shared');
   nav.hidden=true;
   list.querySelectorAll('.archivo,.grupo-pedido').forEach(el=>{el.hidden=false});
   list.querySelectorAll('[data-mobile-hidden]').forEach(el=>{el.dataset.mobileHidden='false'});
   window.parent.postMessage({type:'SABER_MOBILE_STAGE',stage:0},location.origin);
  }
 }

  sendable=event.data.sendable===true;
  price=Number.isSafeInteger(event.data.total)&&event.data.total>0?event.data.total:null;
  if(active()&&stage===6)refresh();
 });
 if(media.addEventListener)media.addEventListener('change',enable);

 enable();

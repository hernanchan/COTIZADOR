function clean(v){return String(v==null?'NO CONSTA — verificar':v)}
function nombreSeguro(v){
 return clean(v).replace(/[\x00-\x1f\x7f]+/g,' ').replace(/\s+/g,' ').trim();
}
function nombreZip(i){return 'PDF_'+String(i+1).padStart(2,'0')+'.pdf'}
function formatoOrden(t,i){
 const medidas=t.formatoFinal==='personalizado_a4'&&Array.isArray(t.medidasOriginalesCm)
  ?t.medidasOriginalesCm.map(p=>({
   pagina:p.pagina,valor:Number(p.anchoCm.toFixed(2))+' × '+
    Number(p.altoCm.toFixed(2))+' cm'
  })):null;
 const unicas=medidas?[...new Set(medidas.map(p=>p.valor))]:[];
 const tamaño=t.formatoFinal==='personalizado_a4'
  ?unicas.length===1?unicas[0]:'Original personalizado (ver medidas por página)'
  :(['a4','a3','a5'].includes(t.formatoFinal)?t.formatoFinal.toUpperCase():null);
 const color=t.modo==='bn'?'Blanco y negro':t.modo==='color'?'Color':null;
 const caras=t.carasPresupuestadas==='simple'?'Simple faz (una cara por hoja)':
  t.carasPresupuestadas==='double'?'Doble faz (ambas caras de la hoja)':null;
 const anillado=t.tipoAnillado==='ninguno'?'Sin anillado':
  t.tipoAnillado==='individual'?'Anillado individual':
  t.tipoAnillado==='grupo'
   ?'Anillado en grupo '+clean(t.grupoAnillado)+' (confirmar integrantes)':null;
 const papeles={
  estandar:'Papel estándar B/N (gramaje sin especificar)',
  obra70:'Papel obra 70 g',obra80:'Papel obra 80 g',
  obra140:'Papel obra 140 g',obra180:'Papel obra 180 g',
  a3_color_obra80:'Papel obra 80 g A3',
  a3_color_obra140:'Papel obra 140 g A3',
  a3_color_obra180:'Papel obra 180 g A3',
  recubierto:'Papel recubierto',fotografico:'Papel fotográfico'
 };
 const papel=papeles[t.material]||t.material||null;
 const elegidas=Array.isArray(t.paginasElegidas)?t.paginasElegidas:[];
 const todas=elegidas.length===t.paginasOriginales&&elegidas.every((v,i)=>v===i+1);
 const paginas=elegidas.length?(todas?'Todas ('+elegidas.length+')':elegidas.join(', ')):null;
 const n=t.paginasPorCaraPresupuestadas;
 const distribucion=[1,2,4,9].includes(n)
  ?n===1?'Normal: 1 página del PDF por cara':
   'Múltiples en Adobe Acrobat: '+n+' páginas del PDF por cara; revisar reducción en vista previa':null;
 const ajuste=n>1?
  'Elegir «Múltiples» en Adobe Acrobat: la escala se adapta a la distribución':
  ['a5','personalizado_a4'].includes(t.formatoFinal)?
  'Tamaño real / 100 %, SIN ampliar el contenido':
  t.formatoFinal==='a4'?'Ajustar proporcionalmente al A4, sin recortar':
  t.formatoFinal==='a3'?'Ajustar proporcionalmente al A3, sin recortar':null;
 const filas=[
  ['Archivo que se abre en Adobe',nombreZip(i)],
  ['Nombre original del cliente',nombreSeguro(t.nombreOriginal)],
  ['Páginas del PDF',Number.isSafeInteger(t.paginasOriginales)?t.paginasOriginales:null],
  ['Páginas a imprimir',paginas],
  ['Tamaño final solicitado',tamaño],
  ...(t.formatoFinal==='personalizado_a4'?
   [['Tarifa aplicada','A4; tamaño original sin ampliar'],
    ...(medidas&&unicas.length>1?
     [['Medidas por página',medidas.map(p=>'pág. '+p.pagina+': '+p.valor).join('; ')]]:[])]:
   t.formatoFinal==='a5'?[['Tarifa aplicada','A4; A5 sin ampliar']]:[]),
  ['Color de impresión',color],
  ['Impresión de caras',caras],
  ['Distribución / páginas por cara',distribucion],
  ['Escala en Adobe Acrobat',ajuste],
  ['Cantidad de copias',t.copias],
  ['Material / papel',papel],
  ['Anillado después de imprimir',anillado],
  ...(t.descripcion?[['Observaciones del pedido',nombreSeguro(t.descripcion)]]:[])
 ];
 if(t.servicio!=='documentos'){
  return [
   ['Archivo que se abre en Adobe',nombreZip(i)],
   ['Nombre original del cliente',nombreSeguro(t.nombreOriginal)],
   ['Servicio',t.servicio],
   ['Páginas seleccionadas',paginas],
   ['Copias',t.copias],
   ['Advertencia','Este ZIP no contiene las instrucciones completas para ploteos u otros servicios. Verificar antes de producir.']
  ];
 }
 return filas;
}
function textoOrden(d){
 const lines=['LIBRERÍA SABER — ORDEN DE TRABAJO N.º '+d.numeroOrden,
  'ESTADO: '+d.estadoPago+' — NO IMPRIMIR NI ENTREGAR SIN AUTORIZACIÓN',
  'Importe del pedido: ARS '+d.importe,
  'Documentos: '+d.trabajos.length,
  'Los PDF originales NO están modificados ni impuestos. Abrir cada PDF en Adobe Acrobat.',
  ''];
 for(const [i,t] of d.trabajos.entries()){
  lines.push('PDF '+String(i+1).padStart(2,'0')+' — '+nombreZip(i)+
   ' — '+nombreSeguro(t.nombreOriginal));
  for(const [k,v] of formatoOrden(t,i))lines.push('  '+k+': '+clean(v));
  lines.push('');
 }
 lines.push('IMPORTANTE: comprobar la vista previa de Adobe, papel y escala antes de imprimir.');
 lines.push('Cualquier «NO CONSTA» requiere verificar la instrucción con el cliente.');
 return lines.join('\n');
}
function escapeHtml(s){return clean(s).replace(/[&<>"']/g,c=>({
 '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'
})[c])}

 const sections=d.trabajos.map((t,i)=>
  '<section><h2>'+escapeHtml(nombreZip(i))+' · '+escapeHtml(nombreSeguro(t.nombreOriginal))+
  '</h2><table>'+formatoOrden(t,i).map(([k,v])=>'<tr><th>'+escapeHtml(k)+
  '</th><td>'+escapeHtml(v)+'</td></tr>').join('')+'</table></section>').join('');
 return '<!doctype html><html lang="es"><head><meta charset="utf-8">'+
  '<title>Orden '+escapeHtml(d.numeroOrden)+'</title><style>'+

  sections+'<p class="nota">Control: verificar vista previa, escala, páginas por cara, doble faz y papel. Si dice NO CONSTA, confirmar antes de imprimir.</p></body></html>';
}
async function fetchFicha(){
 if(!isOrder)throw Error('Referencia del pedido incorrecta');
 const res=await fetch('/prueba-v2/api/ensayo-tus-paquete-manifiesto',{
  method:'POST',credentials:'same-origin',cache:'no-store',
  headers:{'Content-Type':'application/json'},
  body:JSON.stringify({webOrderId})
 });
 const result=await res.json().catch(()=>({}));
 if(!res.ok||result.ok!==true||result.webOrderId!==webOrderId||
    !/^\d{3,}$/.test(result.numeroOrden||'')||
    !Array.isArray(result.trabajos)||!result.trabajos.length||
    result.paymentAuthorized!==false)
  throw Error(result.error||'No se pudo verificar la orden de trabajo');
 return result;
}
async function pdfOriginal(d,t){
 const r=await fetch('/prueba-v2/api/ensayo-tus-descargar-personal',{
  method:'POST',credentials:'same-origin',cache:'no-store',
  headers:{'Content-Type':'application/json'},
  body:JSON.stringify({orderId:d.webOrderId,receiptId:t.reciboPrivado})
 });
 if(!r.ok||!String(r.headers.get('Content-Type')||'').startsWith('application/pdf'))
  throw Error('No se pudo verificar el PDF '+t.orden+' (HTTP '+r.status+')');
 const blob=await r.blob();
 if(blob.size!==t.bytes||blob.size<5||blob.size>20*1024*1024)
  throw Error('Archivo PDF incompleto: '+t.orden);
 const head=new Uint8Array(await blob.slice(0,5).arrayBuffer());
 if(String.fromCharCode(...head)!=='%PDF-')
  throw Error('Contenido PDF inválido: '+t.orden);
 return blob;
}

function rememberImages(files){
 for(const f of [...(files||[])]){
  if(!/^image\/(jpeg|png)$/i.test(f.type)&&!/\.(jpe?g|png)$/i.test(f.name))continue;
  const stem=f.name.replace(/\.(jpe?g|png)$/i,'').replace(/[^\p{L}\p{N}._ -]/gu,'_').slice(0,75);
  if(!stem||thumbnails.has(stem))continue;
  thumbnails.set(stem,null);
  // Miniatura local reducida; no usar blob: como src, porque la CSP comercial
  // puede impedirlo. No transmitir el contenido fuera del dispositivo.
  (async()=>{try{
   const b=await createImageBitmap(f),r=Math.min(1,160/Math.max(b.width,b.height));

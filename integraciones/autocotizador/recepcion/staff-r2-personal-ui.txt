(function(){
 if(location.pathname!=='/personal/paquete')return;
 const section=document.getElementById('r2Inspeccion');
 section.hidden=false;
 const q=id=>document.getElementById(id);
 const order=q('r2Order'),file=q('r2FileId'),status=q('r2Estado');
 const detail=q('r2Detalle'),fic=q('r2Ficha');
 const params=new URL(location.href).searchParams;
 order.value=params.get('webOrderId')||'';
 const selectedFile=params.get('fileId')||'';
 if(/^[A-Za-z0-9_-]{1,80}$/.test(selectedFile))file.value=selectedFile;
 let manifest=null;
 q('r2Recargar').addEventListener('click',async()=>{
  const list=q('r2Lista');list.replaceChildren();
  list.textContent='Consultando pedidos acreditados…';
  try{
   const r=await fetch('/personal/api/live-r2-board',{cache:'no-store'});
   const d=await r.json();
   if(!r.ok||d?.ok!==true||!Array.isArray(d.orders))
    throw Error('La Central no habilitó la consulta de pedidos.');
   list.replaceChildren();
   if(!d.orders.length){list.textContent='Sin pedidos pagados pendientes.';return}
   for(const x of d.orders){
    const item=document.createElement('div');
    item.style.cssText='padding:9px 0;border-bottom:1px solid #dde5ec';
    const label=document.createElement('span');
    const tusCount=Math.max(0,Number(x.fileCount)-Number(x.r2Count));
    label.textContent='Orden '+x.numeroOrden+' · '+x.estado+' · '+x.fileCount+
     ' PDF · TUS '+tusCount+' · R2 '+x.r2Count+
     ' · Inspecciones R2 pendientes '+x.pendingCount+' · $ '+
     Number(x.amountArs).toLocaleString('es-AR');
    item.append(label);
    const use=document.createElement('button');
    use.type='button';
    use.textContent=Number(x.r2Count)===0?'Abrir pedido TUS':
     Number(x.pendingCount)===0?'Abrir ZIP completo':'Inspeccionar PDF R2';
    use.style.marginLeft='10px';
    use.addEventListener('click',()=>{
     if(Number(x.r2Count)===0||Number(x.pendingCount)===0){
      location.assign('/personal/paquete?webOrderId='+encodeURIComponent(x.webOrderId));
      return;
     }
     order.value=x.webOrderId;
     const chosen=Array.isArray(x.stagedR2FileIds)?x.stagedR2FileIds[0]:null;
     file.value=typeof chosen==='string'&&/^[A-Za-z0-9_-]{1,80}$/.test(chosen)?
      chosen:'';
     if(file.value)q('r2Consultar').click();
     section.scrollIntoView({behavior:'smooth'});
    });
    item.append(use);list.append(item);
   }
  }catch(e){list.textContent='No se pudo consultar la lista: '+String(e?.message||e).slice(0,100)}
 });

 const url=()=>new URLSearchParams({webOrderId:order.value.trim(),
  fileId:file.value.trim()}).toString();
 const error=e=>{status.textContent='No se completó: '+String(e?.message||e).slice(0,160)};
 q('r2Consultar').addEventListener('click',async()=>{
  detail.hidden=true;manifest=null;status.textContent='Consultando PDF pendiente…';
  try{
   const response=await fetch('/personal/api/live-r2-inspection-source?'+url(),{cache:'no-store'});
   const d=await response.json();
   if(!response.ok||d?.ok!==true||d.webOrderId!==order.value.trim()||
      d.fileId!==file.value.trim()||!Array.isArray(d.paginasDeclaradas))
    throw Error('La Central no confirmó este PDF pendiente.');
   manifest=d;
   fic.textContent='Archivo: '+d.fileId+' | '+d.bytes.toLocaleString('es-AR')+' bytes | '+
    d.paginasDeclaradas.length+' páginas\nSHA-256: '+d.sha256+
    '\nMedidas cotizadas: '+d.paginasDeclaradas.slice(0,8).map(x=>x.anchoCm+' x '+x.altoCm).join(', ')+
    (d.paginasDeclaradas.length>8?'…':'');
   q('r2Descargar').href='/personal/api/live-r2-inspection-original?'+url();
   const format=String(d.formatoFinal||'').toLowerCase();
   const mixed=new Set(d.paginasDeclaradas.map(x=>
    String(x.anchoCm)+' × '+String(x.altoCm)+' cm')).size;
   q('r2Formato').textContent='PDF original: '+d.paginasDeclaradas.length+
    ' páginas · '+mixed+' tamaño(s) original(es) · impresión solicitada: '+
    (format==='a4'?'A4 (ajustar todas las páginas, sin recortar)':
     format==='a3'?'A3 (ajustar todas las páginas, sin recortar)':
     format==='original'?'tamaño original':
     'consultar ficha de la orden');
   detail.hidden=false;status.textContent='Descargá el original y verificá su contenido. No hace falta transcribir las medidas por página.';
  }catch(e){error(e)}
 });
 q('r2Aprobar').addEventListener('click',async()=>{
  if(!manifest)return;
  status.textContent='Registrando inspección…';
  const button=q('r2Aprobar');button.disabled=true;
  try{
   if(!q('r2Verificado').checked)throw Error('Confirmá la revisión del PDF.');
   const operator=q('r2Operador').value.trim();
   if(operator.length<2||operator.length>60)throw Error('Ingresá el responsable.');
   // Datos originales vinculados a la misma cotización y SHA-256; nunca
   // tomar medidas introducidas manualmente ni asumir PDF uniforme.
   // La inspección visual confirma el contenido, no habilita impresión.
   const measuredDimensions=manifest.paginasDeclaradas.map((p,i)=>({
    numero:i+1,anchoCm:p.anchoCm,altoCm:p.altoCm
   }));
   if(!measuredDimensions.length||measuredDimensions.length>2000||
      measuredDimensions.some(p=>!Number.isFinite(p.anchoCm)||
       !Number.isFinite(p.altoCm)||p.anchoCm<=0||p.altoCm<=0))
    throw Error('La Central no dispone de dimensiones válidas para este PDF.');
   const response=await fetch('/personal/api/live-r2-inspection-confirm',{
    method:'POST',headers:{'Content-Type':'application/json'},
    body:JSON.stringify({webOrderId:manifest.webOrderId,fileId:manifest.fileId,
     operator,confirmed:true,measuredDimensions})});
   const d=await response.json();
   if(!response.ok||d?.ok!==true||d.inspected!==true)
    throw Error('La Central no aprobó la inspección. Revisá geometría y pago.');
   detail.hidden=true;
   status.textContent='Inspección registrada. Actualizando el ZIP y la hoja de instrucciones…';
   setTimeout(()=>location.reload(),1300);
  }catch(e){error(e)}finally{button.disabled=false}
 });
if(file.value&&order.value)queueMicrotask(()=>q('r2Consultar').click());
else if(!order.value)queueMicrotask(()=>q('r2Recargar').click());
})();
